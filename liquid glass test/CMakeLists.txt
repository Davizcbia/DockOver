cmake_minimum_required(VERSION 3.21)
project(LiquidGlassLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2)
qt_add_executable(LiquidGlassLab WIN32 main.cpp)
target_link_libraries(LiquidGlassLab PRIVATE Qt6::Quick Qt6::QuickControls2)
get_filename_component(GLASS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/launcher/project-glass" ABSOLUTE)
target_compile_definitions(LiquidGlassLab PRIVATE GLASS_SOURCE_DIR="${GLASS_SOURCE_DIR}" GLASS_LAB_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
